cmake_minimum_required(VERSION 3.22)
project(SP404_Modular VERSION 0.3.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(SP404_Modular COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.sp404modular" PLUGIN_MANUFACTURER_CODE IndA PLUGIN_CODE S4Md FORMATS VST3 Standalone PRODUCT_NAME "404 Modular MKII")
target_sources(SP404_Modular PRIVATE Source/PluginProcessor.cpp)
juce_generate_juce_header(SP404_Modular)
target_compile_definitions(SP404_Modular PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SP404_Modular PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_processors juce::juce_gui_extra)
